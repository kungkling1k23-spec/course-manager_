        function updateStatistics() {
            const totalCourses = courses.length;
            const totalCredits = courses.reduce((sum, c) => sum + Number(c.credits || 0), 0);
            const totalHours = courses.reduce((sum, c) => sum + Number(c.hours || 0), 0);
            const avgCredits = totalCourses > 0 ? (totalCredits / totalCourses).toFixed(1) : '0';

            document.getElementById('stat-total-courses').innerText = totalCourses;
            document.getElementById('stat-total-credits').innerText = totalCredits;
            document.getElementById('stat-avg-credits').innerText = avgCredits;
            document.getElementById('stat-total-hours').innerText = totalHours;
        }

        // --- FILTER & SORT LOGIC ---
        function applyFilters() {
            const search = document.getElementById('search-input').value.toLowerCase().trim();
            const institution = document.getElementById('filter-institution').value;
            const category = document.getElementById('filter-category').value;
            const sortBy = document.getElementById('sort-by').value;

            filteredCourses = courses.filter(course => {
                const matchSearch = course.code.toLowerCase().includes(search) || 
                                    course.name.toLowerCase().includes(search) ||
                                    (course.institution && course.institution.toLowerCase().includes(search)) ||
                                    (course.onlineSub && course.onlineSub.toLowerCase().includes(search));
                const matchInst = (institution === 'ALL') || (course.institution === institution);
                const matchCat = (category === 'ALL') || (course.category === category);

                return matchSearch && matchInst && matchCat;
            });

            // Sorting
            filteredCourses.sort((a, b) => {
                if (sortBy === 'code') return a.code.localeCompare(b.code);
                if (sortBy === 'name') return a.name.localeCompare(b.name, 'th');
                if (sortBy === 'credits') return b.credits - a.credits;
                if (sortBy === 'hours') return (b.hours || 0) - (a.hours || 0);
                return 0;
            });

            renderTable();
            updateStatistics();
        }

        function handleFormSubmit(e) {

            const code = document.getElementById('form-code').value.trim();
            const name = document.getElementById('form-name').value.trim();
            const institution = document.getElementById('form-institution').value;
            const category = document.getElementById('form-category').value;
            const credits = parseInt(document.getElementById('form-credits').value);
            const hours = parseInt(document.getElementById('form-hours').value);
            const onlineSub = document.getElementById('form-online-sub').value.trim();

            if (id) {
                // Update
                const index = courses.findIndex(c => c.id === id);
                if (index !== -1) {
                    courses[index] = { id, code, name, institution, category, credits, hours, onlineSub };
                    showToast('แก้ไขข้อมูลรายวิชาเรียบร้อย');
                }
            } else {
                // Add New
                const newCourse = {
                    id: 'c-' + Date.now(),
                    code, name, institution, category, credits, hours, onlineSub
                };
                courses.push(newCourse);
                showToast('เพิ่มรายวิชาใหม่สำเร็จ');
            }

            saveCoursesToStorage();
            applyFilters();
            closeModal();

            if (activeDetailCourseId && activeDetailCourseId === id) {
                openDetailDrawer(id);
            }
        }

        function promptDelete(courseId) {
            const course = courses.find(c => c.id === courseId);
            if (!course) return;

            courseToDeleteId = courseId;
            document.getElementById('delete-course-code').innerText = `${course.code} (${course.name})`;
            
            const btn = document.getElementById('confirm-delete-btn');
            btn.onclick = () => confirmDelete();

            const modal = document.getElementById('delete-modal');
            modal.classList.remove('hidden');
            setTimeout(() => {
                modal.classList.remove('opacity-0');

        }

        function confirmDelete() {
            if (!courseToDeleteId) return;
            courses = courses.filter(c => c.id !== courseToDeleteId);
            saveCoursesToStorage();
            applyFilters();
            closeDeleteModal();

                closeDetailDrawer();

            showToast('ลบรายวิชาเรียบร้อยแล้ว', 'trash-2');
        }
